#include <hip/hip_runtime.h>

#define HD 128
#define TILE_ROWS 4

extern "C" __launch_bounds__(32, 8)
__global__ void gated_delta_net_f32(
    const float* __restrict__ q,
    const float* __restrict__ k,
    const float* __restrict__ v,
    const float* __restrict__ gate,
    const float* __restrict__ beta,
    float* __restrict__ state,
    float* __restrict__ output,
    int n_tokens,
    int n_heads,
    int head_dim
) {
    // LANE LAYOUT. The wave is 4 groups of 8 lanes; group `g` owns row
    // row_start+g, lane-in-group `l` owns columns [16l, 16l+16).
    //
    // The previous layout gave every lane 4 columns of ONE row and walked the
    // 4 rows serially, so each row paid two 32-lane tree reductions (5 shuffle
    // steps each, plus a broadcast) back to back: 40+ shuffles per token, all
    // on the critical path. Ablating just those reductions cut t=24 by 45%,
    // which made them the single largest term in the per-token cost.
    //
    // Here the 4 rows are resolved CONCURRENTLY across the wave and each
    // reduction spans only 8 lanes (`width` argument), so a token costs 2x3
    // shuffle steps instead of 4x(5+5). Each lane does 16 columns instead of 4,
    // which is the same total FMA work with 4x the ILP to hide it.
    const int h = blockIdx.x;
    const int tile = blockIdx.y;
    if (h >= n_heads) return;
    const int tid = threadIdx.x;
    const int g = tid >> 3;          // row within the tile
    const int l = tid & 7;           // column group within the row
    const int col = l * 16;
    const int stride = n_heads * HD;
    const int row_start = tile * TILE_ROWS;
    const int row = row_start + g;

    __shared__ float S_tile[TILE_ROWS * HD];

    float* S_global = state + h * HD * HD + row_start * HD;
    for (int i = tid; i < TILE_ROWS * HD; i += 32)
        S_tile[i] = S_global[i];
    __syncthreads();

    // This lane's 16 columns of its own row, held in registers for the whole
    // token loop. Nothing is shared across lanes once loaded.
    float sreg[16];
    #pragma unroll
    for (int i = 0; i < 16; i++) sreg[i] = S_tile[g * HD + col + i];

    for (int t = 0; t < n_tokens; t++) {
        const float* q_t = q + t * stride + h * HD;
        const float* k_t = k + t * stride + h * HD;
        const float* v_t = v + t * stride + h * HD;
        const float alpha = expf(gate[t * n_heads + h]);
        const float beta_v = beta[t * n_heads + h];

        float kk[16], qq[16];
        #pragma unroll
        for (int i = 0; i < 16; i++) { kk[i] = k_t[col + i]; qq[i] = q_t[col + i]; }

        float kv = 0.0f;
        #pragma unroll
        for (int i = 0; i < 16; i++) kv += sreg[i] * kk[i];
        #pragma unroll
        for (int o = 4; o > 0; o >>= 1) kv += __shfl_down(kv, o, 8);
        kv = __shfl(kv, 0, 8);

        const float delta = (v_t[row] - alpha * kv) * beta_v;

        float out_v = 0.0f;
        #pragma unroll
        for (int i = 0; i < 16; i++) {
            sreg[i] = alpha * sreg[i] + kk[i] * delta;
            out_v += sreg[i] * qq[i];
        }
        #pragma unroll
        for (int o = 4; o > 0; o >>= 1) out_v += __shfl_down(out_v, o, 8);
        if (l == 0)
            output[t * stride + h * HD + row] = out_v;
    }

    #pragma unroll
    for (int i = 0; i < 16; i++) S_tile[g * HD + col + i] = sreg[i];
    __syncthreads();
    for (int i = tid; i < TILE_ROWS * HD; i += 32)
        S_global[i] = S_tile[i];
}
